One URL, One Screen.

Angular Routes map browser URLs directly to standalone components. This dedicated screen is rendered inside <router-outlet> at route /routing-example.

Active Route: /routing-example Component: RoutingExampleComponent

Route Configuration (src/app/app.routes.ts)

export const routes: Routes = [
  { path: 'architecture', component: ArchitecturePageComponent },
  { path: 'routing-example', component: RoutingExampleComponent },
  { path: '', pathMatch: 'full', redirectTo: 'architecture' },
  { path: '**', redirectTo: 'architecture' }
];
